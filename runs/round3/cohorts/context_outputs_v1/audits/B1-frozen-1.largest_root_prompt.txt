import re

# Reuse downloads; select verbatim contextual windows from unusually long documents.
def evidence_windows(text, terms, radius=1800, max_chars=32000):
    if len(text) <= max_chars:
        return [text]
    spans = []
    pattern = "|".join(re.escape(term) for term in terms)
    for match in re.finditer(pattern, text, flags=re.I):
        start, end = max(0, match.start() - radius), min(len(text), match.end() + radius)
        if spans and start <= spans[-1][1]:
            spans[-1] = (spans[-1][0], max(end, spans[-1][1]))
        else:
            spans.append((start, end))
    excerpts, used = [], 0
    for start, end in spans:
        if used >= max_chars:
            break
        excerpt = text[start:min(end, start + max_chars - used)]
        excerpts.append(excerpt)
        used += len(excerpt)
    return excerpts
